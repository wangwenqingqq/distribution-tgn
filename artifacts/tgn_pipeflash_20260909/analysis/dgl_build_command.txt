from pathlib import Path
import os,subprocess,json
root=Path("/home/data/wangxuran/tgn_pipeflash_20260909");source=root/'vendor/dgl'
env=dict(os.environ,CUDA_VISIBLE_DEVICES='',CUDACXX='/usr/local/cuda-13.1/bin/nvcc')
cmd=['cmake','-S',str(source),'-B',str(source/'build'),'-DCMAKE_BUILD_TYPE=Release',
 '-DUSE_CUDA=ON','-DCUDA_TOOLKIT_ROOT_DIR=/usr/local/cuda-13.1',
 '-DCUDA_NVCC_EXECUTABLE=/usr/local/cuda-13.1/bin/nvcc','-DCUDA_ARCH_NAME=Manual',
 '-DCUDA_ARCH_BIN=12.0','-DCUDA_ARCH_PTX=','-DUSE_LIBXSMM=OFF','-DBUILD_SPARSE=OFF',
 '-DBUILD_GRAPHBOLT=OFF','-DBUILD_TORCH=OFF','-DBUILD_CPP_TEST=OFF','-DTP_BUILD_TESTING=OFF',
 '-DTP_BUILD_PYTHON=OFF','-DTP_USE_CUDA=OFF','-DCMAKE_CXX_FLAGS=','-DCMAKE_C_FLAGS=-Wno-error=maybe-uninitialized']
(root/'analysis/dgl_build_command.json').write_text(json.dumps(cmd,indent=2)+'\n')
with (root/'logs/dgl_build_v3.log').open('w') as log:
 r=subprocess.run(cmd,env=env,stdout=log,stderr=subprocess.STDOUT);print('configure',r.returncode,flush=True)
 if not r.returncode:
  r=subprocess.run(['cmake','--build',str(source/'build'),'--target','dgl','--parallel','4'],env=env,stdout=log,stderr=subprocess.STDOUT)
  print('build',r.returncode,flush=True)
